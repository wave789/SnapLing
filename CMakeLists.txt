cmake_minimum_required(VERSION 3.22)
project(SnapLing VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
if(UNIX)
 set(CMAKE_INSTALL_RPATH "$ORIGIN/../lib")
endif()
include(CTest)
include(GNUInstallDirs)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network Concurrent Test)
option(SNAPLING_PIPEWIRE "启用 Wayland 即时截图和屏幕共享" ON)
option(SNAPLING_TESSERACT "启用本地 Tesseract OCR" ON)
set(SOURCES
 src/platform/Autostart.cpp src/core/Types.cpp src/capture/CaptureService.cpp src/hotkey/Hotkey.cpp
 src/selection/SelectionOverlay.cpp src/editor/Document.cpp src/editor/Canvas.cpp
 src/editor/EditorSession.cpp src/editor/AnnotationTools.cpp
 src/ocr/OcrEngine.cpp src/ocr/TextLayout.cpp src/translation/TranslationProvider.cpp
 src/image/ImageProcessor.cpp src/history/HistoryStore.cpp src/settings/Settings.cpp
 src/ui/EditorWindow.cpp src/ui/SettingsDialog.cpp src/ui/PinWindow.cpp
 src/app/SingleInstance.cpp src/app/Application.cpp resources/resources.qrc)
if(WIN32)
 list(APPEND SOURCES src/platform/windows/WindowsCapture.cpp)
else()
 find_package(Qt6 6.4 REQUIRED COMPONENTS DBus)
 find_package(PkgConfig REQUIRED)
 find_package(X11 REQUIRED COMPONENTS Xrandr)
 find_library(XCB_LIBRARY xcb REQUIRED)
 list(APPEND SOURCES src/platform/linux/x11/X11Capture.cpp src/platform/linux/wayland/Portal.cpp src/platform/linux/wayland/WaylandCapture.cpp)
 if(SNAPLING_PIPEWIRE)
  pkg_check_modules(PIPEWIRE IMPORTED_TARGET libpipewire-0.3)
  if(PIPEWIRE_FOUND)
   list(APPEND SOURCES src/platform/linux/wayland/PipeWireStream.cpp)
  else()
   message(WARNING "未找到 PipeWire 开发库。Wayland 即时框选需要 libpipewire-0.3-dev；此构建仅尝试非交互式 Screenshot Portal。")
  endif()
 endif()
endif()
add_library(snapling_lib STATIC ${SOURCES})
target_include_directories(snapling_lib PUBLIC src)
target_link_libraries(snapling_lib PUBLIC Qt6::Widgets Qt6::Network Qt6::Concurrent)
if(MSVC)
 target_compile_options(snapling_lib PUBLIC /utf-8 /W4)
else()
 target_compile_options(snapling_lib PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
 target_link_libraries(snapling_lib PUBLIC user32 gdi32 dwmapi)
 target_compile_definitions(snapling_lib PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
else()
 target_link_libraries(snapling_lib PUBLIC Qt6::DBus X11::X11 X11::Xrandr ${XCB_LIBRARY})
 if(SNAPLING_PIPEWIRE AND PIPEWIRE_FOUND)
  target_link_libraries(snapling_lib PRIVATE PkgConfig::PIPEWIRE)
  target_compile_definitions(snapling_lib PUBLIC SNAPLING_HAVE_PIPEWIRE)
 endif()
endif()
if(SNAPLING_TESSERACT)
 if(WIN32)
  find_package(Tesseract CONFIG REQUIRED)
  target_link_libraries(snapling_lib PRIVATE Tesseract::libtesseract)
 else()
  pkg_check_modules(TESSERACT REQUIRED IMPORTED_TARGET tesseract)
  target_link_libraries(snapling_lib PRIVATE PkgConfig::TESSERACT)
 endif()
 target_compile_definitions(snapling_lib PUBLIC SNAPLING_HAVE_TESSERACT)
endif()
if(NOT WIN32)
 pkg_check_modules(ZXING IMPORTED_TARGET zxing)
 if(ZXING_FOUND)
  target_link_libraries(snapling_lib PRIVATE PkgConfig::ZXING)
  target_compile_definitions(snapling_lib PUBLIC SNAPLING_HAVE_ZXING)
 endif()
endif()
add_executable(snapling WIN32 src/app/main.cpp)
target_link_libraries(snapling PRIVATE snapling_lib)
if(WIN32)
 target_sources(snapling PRIVATE packaging/windows/app.rc packaging/windows/app.manifest)
 target_include_directories(snapling PRIVATE packaging/windows resources)
endif()
install(TARGETS snapling RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/linux/io.snapling.SnapLing.desktop DESTINATION share/applications)
install(FILES resources/snapling.svg DESTINATION share/icons/hicolor/scalable/apps)
install(FILES README.md ARCHITECTURE.md BUILD.md TESTING.md LIMITATIONS.md THIRD_PARTY.md LICENSE DESTINATION share/doc/snapling)
install(DIRECTORY docs DESTINATION share/doc/snapling)
install(FILES models/README.md models/manifest.json models/TESSDATA-LICENSE models/SHA256SUMS DESTINATION share/doc/snapling/models)
install(DIRECTORY models/tessdata DESTINATION share/snapling)
install(FILES models/TESSDATA-LICENSE models/SHA256SUMS models/manifest.json DESTINATION share/snapling)
if(BUILD_TESTING)
 add_executable(snapling_tests tests/TestCore.cpp)
 target_link_libraries(snapling_tests PRIVATE snapling_lib Qt6::Test)
 add_test(NAME core COMMAND snapling_tests)
 set_tests_properties(core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;TESSDATA_PREFIX=${PROJECT_SOURCE_DIR}/models/tessdata")
 if(UNIX)
  add_executable(snapling_portal_test tests/TestPortal.cpp)
  target_link_libraries(snapling_portal_test PRIVATE snapling_lib Qt6::Test)
  add_executable(snapling_capture_flow_test tests/TestCaptureFlow.cpp)
  target_link_libraries(snapling_capture_flow_test PRIVATE snapling_lib Qt6::Test)
  find_program(DBUS_RUN_SESSION dbus-run-session)
  if(DBUS_RUN_SESSION)
   add_test(NAME capture_flow COMMAND ${DBUS_RUN_SESSION} -- $<TARGET_FILE:snapling_capture_flow_test>)
  else()
   message(STATUS "Wayland capture-flow tests require dbus-run-session")
  endif()
 endif()
 add_executable(snapling_desktop_test tests/TestDesktop.cpp)
 target_link_libraries(snapling_desktop_test PRIVATE snapling_lib Qt6::Test)
 target_compile_definitions(snapling_desktop_test PRIVATE SNAPLING_TEST_MODEL_DIR="${PROJECT_SOURCE_DIR}/models/tessdata")
 if(UNIX)
  add_executable(snapling_quick_edit_test tests/TestQuickEdit.cpp)
  target_link_libraries(snapling_quick_edit_test PRIVATE snapling_lib Qt6::Test)
  add_dependencies(snapling_quick_edit_test snapling_desktop_test)
  find_program(XVFB_RUN xvfb-run)
  find_program(XDOTOOL xdotool)
  if(XVFB_RUN AND XDOTOOL)
   add_test(NAME quick_edit COMMAND ${XVFB_RUN} -a -s "-screen 0 1920x1080x24" $<TARGET_FILE:snapling_quick_edit_test>)
   set_tests_properties(quick_edit PROPERTIES ENVIRONMENT "TESSDATA_PREFIX=${PROJECT_SOURCE_DIR}/models/tessdata")
   foreach(SCALE IN ITEMS 1.5 2)
    add_test(NAME quick_edit_scale_${SCALE} COMMAND ${XVFB_RUN} -a -s "-screen 0 1920x1080x24" $<TARGET_FILE:snapling_quick_edit_test>
     instantAnnotations nativeMouseRelease instantOcr advancedHandoff translatedPng)
    set_tests_properties(quick_edit_scale_${SCALE} PROPERTIES ENVIRONMENT "TESSDATA_PREFIX=${PROJECT_SOURCE_DIR}/models/tessdata;QT_SCALE_FACTOR=${SCALE}")
   endforeach()
   # xvfb-run -a 在并行启动时可能同时选到同一空闲显示号。
   set_tests_properties(quick_edit quick_edit_scale_1.5 quick_edit_scale_2
    PROPERTIES RESOURCE_LOCK snapling_xvfb TIMEOUT 180)
  endif()
 endif()
endif()
set(CPACK_PACKAGING_INSTALL_PREFIX "/usr")
set(CPACK_STRIP_FILES TRUE)
set(CPACK_PACKAGE_NAME snapling)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_CONTACT "SnapLing contributors")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "qt6-qpa-plugins, qt6-wayland, fonts-noto-cjk")
set(CPACK_NSIS_DISPLAY_NAME "SnapLing")
include(CPack)
